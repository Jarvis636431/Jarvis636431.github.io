1.手写日志上报

```javascript
题目：
 现在有一个基于 http 的前端日志采集系统，为了优化性能，我们需要把较密集的单次上报日志的行为，
 改成合并掉 100ms 时间窗口内的单次上报为批量上报，服务端已支持一次请求带多个 Log 数据
 输入输出样例
 1. 没有日志上报调用时，不发生 http 请求。
 2. 无论日志上报调用有多频繁，http 请求间隔不小于 100ms。
 // 已有的上报日志工具函数，已支持批量上报日志
 function uploadLog(logs: string[]): void;
 请实现此函数：
 function reportLog(logStr) {
 // 补全这个方法
 }

```

###### 答案：

```javascript
// 日志缓冲区
let buffer: string[] = [];
// 定时器引用
let timer: ReturnType<typeof setTimeout> | null = null;
// 已有工具：支持批量发送
declare function uploadLog(logs: string[]): void;
function reportLog(logStr: string) {
  // 写入缓冲
  buffer.push(logStr);
  // 如果当前没有发送计划，则启动 100ms 定时器
  if (!timer) {
    timer = setTimeout(() => {
      // 仅在有日志时发送
      if (buffer.length > 0) {
        uploadLog(buffer);
        buffer = [];
      }
      timer = null; // 重置，允许下一轮窗口开始
    }, 100);
  }
}

```

2.手写 Event 事件发布订阅模型

```javascript
题目：实现一个事件收发器 Event 类，继承自此类的对象拥有 on,off,once 和 trigger 方法，用TS编写。
 示例：const event = new Event();
 const log = (val) => console.log(val)
 event.on('event', log);
 event.once('event', log);
 event.trigger('event', 'abc'); // abc, abc
 event.trigger('event', 'abc'); // abc
 event.off('event', log);
 event.trigger('event', 'abc'); // undefined

```

```javascript
class Event {
  constructor() {
    this.events = new Map();
    this.onceMap = new WeakMap(); // <originalHandler, wrapper>
  }
  on(event, handler) {
    const handlers = this.events.get(event) || [];
    handlers.push(handler);
    this.events.set(event, handlers);
  }
  once(event, handler) {
    const wrapper = (...args) => 
      handler(...args);
      this.off(event, handler); // 注意这里仍然传原 handler
    };
    this.onceMap.set(handler, wrapper);
    this.on(event, wrapper);
  }
  off(event, handler) {
    const handlers = this.events.get(event);
    if (!handlers) return;
    // 如果是 once 注册的，找到对应 wrapper
    const wrapped = this.onceMap.get(handler);
    const target = wrapped || handler;
    const filtered = handlers.filter(h => h !== target);
    if (filtered.length > 0) {
      this.events.set(event, filtered);
    } else {
      this.events.delete(event);
    }
    // 清理映射
    if (wrapped) {
      this.onceMap.delete(handler);
    }
  }
  trigger(event, ...args) {
    const handlers = this.events.get(event);
    if (!handlers) return;
    handlers.slice().forEach(h => h(...args));
  }
}

```
